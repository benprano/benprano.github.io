\cvproject{TF-JEPA, FASTER \& DyFAIP — joint imputation and prediction}{2025 – 2026}{JEPA, diffusion, state transitions, irregular time series}
\begin{cvitems}
  \item \textbf{TF-JEPA}: time-frequency joint-embedding predictive architecture for structure-guided imputation of missing values.
  \item \textbf{FASTER \& DyFAIP}: frequency-conditioned diffusion and state-transition models for joint imputation and trajectory prediction on highly sparse clinical timelines.
\end{cvitems}

\cvproject{Drone micro-target detection — RT-DETRv2 with a DINOv2 backbone}{2025 – 2026}{RT-DETRv2, DINOv2 ViT-B, SAHI/ASAHI, false-negative audit}
\begin{cvitems}
  \item Integrated a self-supervised DINOv2 ViT-Base backbone (768-dim) with multi-scale projection (P3–P5) into RT-DETRv2 to preserve representations of targets only a few pixels wide.
  \item Two-pass inference on 20 MP images: a global 3200 $\times$ 3200 scan followed by dynamic SAHI tiling with Greedy NMM merging.
  \item Post-validation audit isolating misses by IoU, with confusion matrices adapted to rare-class oversampling.
\end{cvitems}

\cvproject{Drone small-object detection with Top-K spatial attention}{2025 – 2026}{YOLOv8 / YOLO26-P2, Top-K attention, DBSCAN, Grad-CAM, CLAHE}
\begin{cvitems}
  \item Designed a dynamic spatial-attention projector (ActiveTopKFeatureProjector) that re-injects high-resolution P2/P3 activation maps into the detection head.
  \item Multi-process tiling guided by DBSCAN clustering, CLAHE contrast enhancement and adaptive oversampling of rare species.
  \item Multi-scale SAHI/ASAHI sliced inference and multi-level Grad-CAM to track saliency retention through deep layers.
\end{cvitems}

\cvproject{Multivariate time-series anomaly detection}{2025 – 2026}{MTGNN, GNN-LSTM, SWaT, WADI, TensorBoard}
\begin{cvitems}
  \item Implemented and evaluated graph-based detectors (MTGNN, GNN-LSTM) on multivariate industrial data (SWaT, WADI), with a complete training, evaluation and visualisation framework.

\cvproject{Conversational commerce agent — hybrid RAG and tool execution}{2026}{BM25 + dense, RRF, cross-encoder, Pydantic, DeepEval, Gradio}
